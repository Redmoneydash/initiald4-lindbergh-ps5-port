cmake_minimum_required(VERSION 3.20)
project(letsgojungle VERSION 0.1.0 LANGUAGES C)

# Let's Go Jungle (Sega, 2006), statically recompiled.
#
# The game target only exists once you have lifted an ELF, because the runtime
# needs two headers generated from it. Without them this still configures and
# builds the toolkit's own self-check, so the repo is testable out of the box.
#
#   py -3.11 -m tools recomp main.elf ..\games\letsgojungle\generated
#   cmake -S . -B build -A Win32
#   cmake --build build --config Release

add_subdirectory(lindberghrecomp)

set(GEN "${CMAKE_CURRENT_SOURCE_DIR}/games/letsgojungle/generated")

set(GEN4 "${CMAKE_CURRENT_SOURCE_DIR}/games/initiald4/generated")
file(GLOB LIFTED "${GEN4}/recomp_funcs_*.c")
if(LIFTED AND EXISTS "${GEN4}/recomp_imports.h")
    add_executable(id4
        games/initiald4/src/host.c
        games/initiald4/src/linux_stubs.c
        ${LIFTED}
    )
    lindbergh_runtime(id4 "${GEN4}")
    if(NOT WIN32)
        # Linux desktop reference build: the Orbis-cut window/GL files drive a
        # real headless EGL context (llvmpipe pbuffer) so the game's emitted
        # frames can be compared against the PS5 output pixel-for-pixel.
        target_sources(id4 PRIVATE
            "${LINDBERGH_RUNTIME_DIR}/hle_window_orbis.c"
            "${LINDBERGH_RUNTIME_DIR}/hle_gl_orbis.c"
            "${CMAKE_CURRENT_SOURCE_DIR}/games/initiald4/src/hle_host_linux.c"
            "${CMAKE_CURRENT_SOURCE_DIR}/games/initiald4/src/libsegaapi_linux.c")
        target_compile_definitions(id4 PRIVATE LINUX_EGL)
        target_link_libraries(id4 PRIVATE m EGL GL dl pthread X11 asound)
    endif()
    message(STATUS "id4: building against lifted code in ${GEN4}")
else()
    message(STATUS
        "letsgojungle: no lifted code in ${GEN} yet, building the toolkit "
        "self-check only. See the README.")
endif()
